CREATE DATABASE IF NOT EXISTS `db_inventaris_lab` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE `db_inventaris_lab`;

-- 1. TABEL USER
CREATE TABLE IF NOT EXISTS `user` (
  `id_user` INT AUTO_INCREMENT PRIMARY KEY,
  `nama_lengkap` VARCHAR(255) NOT NULL,
  `email` VARCHAR(255) UNIQUE NOT NULL,
  `no_telp` VARCHAR(20) DEFAULT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 2. TABEL LOGIN
CREATE TABLE IF NOT EXISTS `login` (
  `id_login` INT AUTO_INCREMENT PRIMARY KEY,
  `id_user` INT NOT NULL,
  `username` VARCHAR(255) UNIQUE NOT NULL,
  `password` VARCHAR(255) NOT NULL,
  `role` ENUM('Admin Lab', 'Kepala Lab') NOT NULL,
  FOREIGN KEY (`id_user`) REFERENCES `user`(`id_user`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 3. TABEL DATA MASTER ALAT
CREATE TABLE IF NOT EXISTS `data_master_alat` (
  `id_alat` VARCHAR(50) PRIMARY KEY,
  `nama_alat` VARCHAR(255) NOT NULL,
  `kode_alat` VARCHAR(50) NOT NULL,
  `asal_negara` VARCHAR(100) NOT NULL,
  `harga_satuan` DECIMAL(15,2) NOT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 4. TABEL KONDISI & INVENTARISIR (AUDIT TRAIL)
CREATE TABLE IF NOT EXISTS `kondisi_inventarisir` (
  `id_inventaris` INT AUTO_INCREMENT PRIMARY KEY,
  `id_alat` VARCHAR(50) NOT NULL,
  `tanggal_cek` DATE NOT NULL,
  `kondisi` ENUM('Baik', 'Rusak') NOT NULL,
  `keterangan` TEXT,
  `id_user` INT NOT NULL,
  FOREIGN KEY (`id_alat`) REFERENCES `data_master_alat`(`id_alat`) ON DELETE CASCADE,
  FOREIGN KEY (`id_user`) REFERENCES `user`(`id_user`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 5. TABEL LAPORAN NILAI ASET
CREATE TABLE IF NOT EXISTS `laporan_nilai_aset` (
  `id_laporan` INT AUTO_INCREMENT PRIMARY KEY,
  `periode` VARCHAR(100) NOT NULL,
  `total_jumlah_alat` INT NOT NULL,
  `total_nilai_aset` DECIMAL(15,2) NOT NULL,
  `status_validasi` ENUM('Menunggu Validasi', 'Divalidasi Kepala Lab') DEFAULT 'Menunggu Validasi',
  `divalidasi_oleh` INT DEFAULT NULL,
  `tanggal_validasi` DATETIME DEFAULT NULL,
  FOREIGN KEY (`divalidasi_oleh`) REFERENCES `user`(`id_user`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

<?php
// auth/proses_login.php - Validasi Login & Password Hash PDO
session_start();
require_once '../config/database.php';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = trim($_POST['username']);
    $password = $_POST['password'];

    // Prepared statement untuk cegah SQL Injection
    $stmt = $pdo->prepare("SELECT l.*, u.nama_lengkap FROM login l JOIN user u ON l.id_user = u.id_user WHERE l.username = :username");
    $stmt->execute(['username' => $username]);
    $user = $stmt->fetch();

    if ($user && password_verify($password, $user['password'])) {
        $_SESSION['id_user']      = $user['id_user'];
        $_SESSION['username']     = $user['username'];
        $_SESSION['nama_lengkap'] = $user['nama_lengkap'];
        $_SESSION['role']         = $user['role'];

        header("Location: ../dashboard.php");
        exit;
    } else {
        $_SESSION['error'] = "Username atau password salah!";
        header("Location: login.php");
        exit;
    }
}
?>
Panduan Instalasi XAMPP & Poin Penjelasan Sidang TA
  1. Instalasi XAMPP: Buka XAMPP Control Panel, jalankan Apache dan MySQL.
  2. Import Database: Buka http://localhost/phpmyadmin, buat database bernama db_inventaris_lab, lalu import file SQL di Tab 1.
  3. Struktur Folder: Buat folder inventaris-lab di dalam htdocs/ XAMPP dan salin file-file PHP di atas.
  4. Penjelasan Pengendalian Internal saat Sidang:
    • Sistem mengimplementasikan Pemisahan Tugas (Segregation of Duties): Petugas Admin Lab melakukan pencatatan inventarisir fisik, sedangkan Kepala Lab memegang hak otorisasi validasi nilai aset.
    • Sistem mencatat Jejak Audit (Audit Trail) pada tabel kondisi_inventarisir yang menghubungkan foreign key id_user secara permanen tanpa menimpa histori lama.
